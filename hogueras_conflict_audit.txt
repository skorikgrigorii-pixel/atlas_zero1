   46: class StoryEngine:
   47:     """Alpha 2.3 Story Engine.
   48: 
   49:     Builds a real scene structure and a shot plan instead of a flat synthetic list.
   50:     The engine uses the locked Franklin narrative blueprint, distributes shot timing
   51:     across scenes, rotates visual needs and camera motion, and writes scenes + shots
   52:     to SQLite so Director AI and Timeline can operate on a structured film model.
   53:     """
   54: 
   55:     def __init__(self, db: Database, project_id: str = 'franklin'):
   56:         self.db = db
   57:         self.project_id = project_id
   58:         self.bus = EventBus(db, project_id)
   59:         self._ensure_schema()
   60: 
   61:     def _ensure_schema(self) -> None:
   62:         self.db.conn.executescript('''
   63:         CREATE TABLE IF NOT EXISTS story_scenes(
   64:           id TEXT PRIMARY KEY,
   65:           project_id TEXT NOT NULL,
   66:           idx INTEGER NOT NULL,
   67:           block TEXT NOT NULL,
   68:           title TEXT NOT NULL,
   69:           start_sec REAL NOT NULL,
   70:           end_sec REAL NOT NULL,
   71:           duration_sec REAL NOT NULL,
   72:           target_shots INTEGER NOT NULL,
   73:           narrative_goal TEXT,
   74:           emotional_goal TEXT,
   75:           visual_strategy TEXT,
   76:           required_assets TEXT,
   77:           status TEXT DEFAULT 'planned',
   78:           coverage REAL DEFAULT 0,
   79:           created_at TEXT DEFAULT CURRENT_TIMESTAMP
   80:         );
   81:         ''')
   82:         self.db.conn.commit()
   83: 
   84:     def build_shots(self, target_count: int = 165, duration_sec: float = 960.0) -> dict[str, Any]:
   85:         self.db.execute('DELETE FROM shots WHERE project_id=?', (self.project_id,))
   86:         self.db.execute('DELETE FROM story_scenes WHERE project_id=?', (self.project_id,))
   87:         scene_counts = self._allocate_shots(target_count)
   88:         current_time = 0.0
   89:         shot_idx = 1
   90:         scene_rows: list[tuple[Any, ...]] = []
   91:         shot_rows: list[tuple[Any, ...]] = []
   92:         for scene_idx, bp in enumerate(SCENE_BLUEPRINTS, start=1):
   93:             count = scene_counts[scene_idx - 1]
   94:             scene_duration = round(duration_sec * (bp.duration_sec / self._blueprint_duration()), 3)
   95:             start = round(current_time, 3)
   96:             end = round(current_time + scene_duration, 3)
   97:             scene_rows.append((
   98:                 bp.scene_id, self.project_id, scene_idx, bp.block, bp.title, start, end, scene_duration, count,
   99:                 bp.narrative_goal, bp.emotional_goal, bp.visual_strategy, json.dumps(list(bp.visual_needs), ensure_ascii=False), 'planned', 0.0
